[OxbowUI](https://oxbowui.com) is an open-source component library with 400+ blocks built with Tailwind CSS. This tutorial shows how to use OxbowUI's ecommerce patterns with CoCart data to build product pages in Astro.

## What you'll build

- A responsive product grid for listing products
- A detailed product page with image, price, and description
- A variable product page with image gallery, color/size selectors, and accordion sections
- Add-to-cart functionality using a React island

## Prerequisites

- An Astro project [set up with CoCart](./setup)
- CoCart installed and activated on your WordPress site
- Basic knowledge of Astro and Tailwind CSS

<Note>
  If you haven't set up your Astro project yet, follow the [Astro Setup Guide](./setup) first. This tutorial assumes you have the CoCart API client (`src/lib/cocart.ts`) and base layout from that guide.
</Note>

## Product Grid component

Create `src/components/ProductGrid.astro`:

```astro
---
interface Props {
  products: any[];
}

const { products } = Astro.props;

function formatPrice(price: string, symbol = '$') {
  return `${symbol}${(parseInt(price) / 100).toFixed(2)}`;
}
---

<div class="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
  {products.map((product) => (
    <a href={`/product/${product.slug}`} class="group flex flex-col gap-2">
      <div class="relative aspect-[4/3] overflow-hidden rounded-2xl bg-zinc-100">
        <img
          src={product.images?.[0]?.src || '/placeholder.jpg'}
          alt={product.images?.[0]?.alt || product.name}
          class="absolute inset-0 w-full h-full object-cover transition-transform duration-300 group-hover:scale-105"
          loading="lazy"
        />
      </div>
      <div class="flex items-center justify-between w-full">
        <h3 class="text-sm font-medium text-zinc-900 dark:text-white">
          {product.name}
        </h3>
        <span class="text-sm text-zinc-500 dark:text-zinc-300">
          {formatPrice(product.prices.price, product.prices.currency_symbol)}
        </span>
      </div>
    </a>
  ))}
</div>
```

<Note>
  CoCart returns prices in the smallest currency unit (cents). The `formatPrice` function divides by 100 to display correctly. Adjust this if your store uses a different format.
</Note>

## Variable Product component

For products with variations (colors, sizes), create `src/components/VariableProduct.tsx` as a React island:

```tsx
import { useState } from 'react';

interface VariableProductProps {
  product: any;
}

function formatPrice(price: string, symbol = '$') {
  return `${symbol}${(parseInt(price) / 100).toFixed(2)}`;
}

export default function VariableProduct({ product }: VariableProductProps) {
  const [activeImage, setActiveImage] = useState(0);
  const [activeColor, setActiveColor] = useState<string | null>(null);
  const [activeSize, setActiveSize] = useState<string | null>(null);
  const [adding, setAdding] = useState(false);
  const [message, setMessage] = useState('');

  const images = product.images || [];
  const colorAttr = product.attributes?.find(
    (a: any) => a.name === 'Color' || a.slug === 'pa_color'
  );
  const sizeAttr = product.attributes?.find(
    (a: any) => a.name === 'Size' || a.slug === 'pa_size'
  );

  async function handleAddToCart() {
    if (colorAttr && !activeColor) {
      setMessage('Please select a color');
      return;
    }
    if (sizeAttr && !activeSize) {
      setMessage('Please select a size');
      return;
    }

    setAdding(true);
    setMessage('');

    try {
      const response = await fetch('/api/cart/add', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          id: String(product.id),
          quantity: 1,
          variation: {
            ...(activeColor && { attribute_pa_color: activeColor }),
            ...(activeSize && { attribute_pa_size: activeSize }),
          },
        }),
      });

      if (response.ok) {
        setMessage('Added to cart!');
      } else {
        const error = await response.json();
        setMessage(error.error || 'Failed to add to cart');
      }
    } catch {
      setMessage('Network error');
    } finally {
      setAdding(false);
      setTimeout(() => setMessage(''), 3000);
    }
  }

  return (
    <div className="grid grid-cols-1 lg:grid-cols-2 gap-8">
      {/* Image gallery */}
      <div className="flex flex-col lg:sticky lg:top-24 lg:self-start gap-2">
        <div className="overflow-hidden aspect-square bg-zinc-100 dark:bg-zinc-800 rounded-2xl">
          {images[activeImage] && (
            <img
              src={images[activeImage].src}
              alt={images[activeImage].alt || product.name}
              className="w-full h-full object-cover"
            />
          )}
        </div>

        {images.length > 1 && (
          <div className="grid grid-cols-6 gap-2">
            {images.map((img: any, i: number) => (
              <button
                key={i}
                onClick={() => setActiveImage(i)}
                className={`overflow-hidden aspect-square bg-zinc-100 rounded-xl ${
                  activeImage === i ? 'ring-2 ring-zinc-900 dark:ring-zinc-100' : ''
                }`}
              >
                <img
                  src={img.src}
                  alt={img.alt || 'Thumbnail'}
                  className="w-full h-full object-cover"
                />
              </button>
            ))}
          </div>
        )}
      </div>

      {/* Product details */}
      <div className="flex flex-col">
        <div className="flex items-center justify-between">
          <h1 className="text-xl md:text-2xl font-medium text-zinc-900 dark:text-white">
            {product.name}
          </h1>
          <p className="text-lg text-zinc-900 dark:text-white">
            {formatPrice(product.prices.price, product.prices.currency_symbol)}
          </p>
        </div>

        {(product.short_description || product.description) && (
          <p
            className="text-sm mt-4 text-zinc-500 dark:text-zinc-300"
            dangerouslySetInnerHTML={{
              __html: product.short_description || product.description,
            }}
          />
        )}

        {/* Color selector */}
        {colorAttr && (
          <div className="mt-6">
            <p className="text-xs uppercase text-zinc-500 dark:text-zinc-300 mb-2">
              Color{activeColor ? `: ${activeColor}` : ''}
            </p>
            <div className="flex flex-wrap gap-3">
              {colorAttr.options.map((color: string) => (
                <button
                  key={color}
                  onClick={() => setActiveColor(color)}
                  aria-label={color}
                  className={`w-7 h-7 rounded-full border-2 transition-all ${
                    activeColor === color
                      ? 'border-zinc-900 dark:border-zinc-100 ring-2 ring-offset-2 ring-zinc-900 dark:ring-zinc-100'
                      : 'border-zinc-300'
                  }`}
                  style={{ backgroundColor: color.toLowerCase() }}
                />
              ))}
            </div>
          </div>
        )}

        {/* Size selector */}
        {sizeAttr && (
          <div className="mt-6">
            <p className="text-xs uppercase text-zinc-500 dark:text-zinc-300 mb-2">Size</p>
            <div className="grid grid-cols-4 gap-2">
              {sizeAttr.options.map((size: string) => (
                <button
                  key={size}
                  onClick={() => setActiveSize(size)}
                  className={`flex items-center justify-center px-3 py-2 text-sm font-medium rounded-md transition-all ${
                    activeSize === size
                      ? 'ring-2 ring-zinc-900 dark:ring-zinc-100 text-zinc-900 dark:text-white'
                      : 'ring-1 ring-zinc-200 dark:ring-zinc-700 text-zinc-500 dark:text-zinc-300'
                  }`}
                >
                  {size}
                </button>
              ))}
            </div>
          </div>
        )}

        {/* Status message */}
        {message && (
          <p className={`text-sm mt-4 ${
            message.includes('Added') ? 'text-green-600' : 'text-red-500'
          }`}>
            {message}
          </p>
        )}

        {/* Action buttons */}
        <div className="flex gap-2 mt-8">
          <button
            onClick={handleAddToCart}
            disabled={adding}
            className="flex-1 inline-flex items-center justify-center font-medium rounded-md text-white bg-zinc-900 hover:bg-zinc-950 dark:bg-zinc-100 dark:text-zinc-900 dark:hover:bg-zinc-200 h-9 px-4 text-sm transition-colors disabled:opacity-50"
          >
            {adding ? 'Adding...' : 'Add to Cart'}
          </button>
          <button className="flex-1 inline-flex items-center justify-center font-medium rounded-md text-zinc-600 bg-zinc-50 hover:bg-zinc-100 dark:text-zinc-100 dark:bg-zinc-800 dark:hover:bg-zinc-700 ring-1 ring-zinc-200 dark:ring-zinc-700 h-9 px-4 text-sm transition-colors">
            Buy Now
          </button>
        </div>

        <p className="text-sm mt-2 text-zinc-500 dark:text-zinc-300">
          Free shipping over $50
        </p>

        {/* Accordion sections */}
        <div className="mt-8 border-y border-zinc-200 dark:border-zinc-700 divide-y divide-zinc-200 dark:divide-zinc-700">
          <details>
            <summary className="flex items-center justify-between py-4 text-sm font-medium text-zinc-900 dark:text-white cursor-pointer select-none">
              Details
              <PlusIcon />
            </summary>
            <div className="pb-4">
              <p
                className="text-sm text-zinc-500 dark:text-zinc-300"
                dangerouslySetInnerHTML={{ __html: product.description }}
              />
            </div>
          </details>

          <details>
            <summary className="flex items-center justify-between py-4 text-sm font-medium text-zinc-900 dark:text-white cursor-pointer select-none">
              Shipping
              <PlusIcon />
            </summary>
            <div className="pb-4">
              <p className="text-sm text-zinc-500 dark:text-zinc-300">
                Free standard shipping on orders above $50. Express options available at checkout.
              </p>
            </div>
          </details>

          <details>
            <summary className="flex items-center justify-between py-4 text-sm font-medium text-zinc-900 dark:text-white cursor-pointer select-none">
              Returns
              <PlusIcon />
            </summary>
            <div className="pb-4">
              <p className="text-sm text-zinc-500 dark:text-zinc-300">
                Returns accepted within 30 days of purchase. Items must be in original condition.
              </p>
            </div>
          </details>
        </div>
      </div>
    </div>
  );
}

function PlusIcon() {
  return (
    <svg
      xmlns="http://www.w3.org/2000/svg"
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      className="w-4 h-4"
    >
      <path d="M12 5v14" />
      <path d="M5 12h14" />
    </svg>
  );
}
```

<Tip>
  Install the React integration if you haven't already: `npx astro add react`. This lets Astro hydrate React components as interactive islands.
</Tip>

## Product detail page

Create `src/pages/product/[slug].astro` to handle both simple and variable products:

```astro
---
import Layout from '../../layouts/Layout.astro';
import ProductDetailComponent from '../../components/ProductDetail.astro';
import VariableProductComponent from '../../components/VariableProduct';
import { getProducts } from '../../lib/cocart';

const { slug } = Astro.params;

const products = await getProducts({ slug });
const product = products[0];

if (!product) {
  return Astro.redirect('/404');
}

const isVariable = product.type === 'variable';
---

<Layout title={product.name} description={product.short_description}>
  <main class="container mx-auto px-4 py-12">
    {isVariable ? (
      <VariableProductComponent product={product} client:load />
    ) : (
      <ProductDetailComponent product={product} />
    )}
  </main>
</Layout>
```

The `client:load` directive tells Astro to hydrate the React component on the client, making the color/size selectors and add-to-cart button interactive.

## Cart API endpoint

Create `src/pages/api/cart/add.ts` to proxy add-to-cart requests through your server:

```typescript
import type { APIRoute } from 'astro';
import { addToCart } from '../../../lib/cocart';

export const POST: APIRoute = async ({ request }) => {
  try {
    const { id, quantity = 1, ...options } = await request.json();
    const result = await addToCart(id, quantity, options);

    return new Response(JSON.stringify(result), {
      status: 200,
      headers: { 'Content-Type': 'application/json' },
    });
  } catch (error) {
    return new Response(
      JSON.stringify({ error: (error as Error).message }),
      {
        status: 500,
        headers: { 'Content-Type': 'application/json' },
      }
    );
  }
};
```

## Add-to-cart for simple products

The ProductDetail component uses a `data-action="add-to-cart"` attribute on the button. Add a small script to your layout or page to handle clicks:

```astro
<script>
  document.addEventListener('click', async (e) => {
    const button = (e.target as HTMLElement).closest('[data-action="add-to-cart"]');
    if (!button) return;

    const productId = button.getAttribute('data-product-id');
    if (!productId) return;

    button.textContent = 'Adding...';
    button.setAttribute('disabled', 'true');

    try {
      const res = await fetch('/api/cart/add', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id: productId, quantity: 1 }),
      });

      button.textContent = res.ok ? 'Added!' : 'Error';
    } catch {
      button.textContent = 'Error';
    }

    setTimeout(() => {
      button.textContent = 'Add to Cart';
      button.removeAttribute('disabled');
    }, 2000);
  });
</script>
```

This uses vanilla JavaScript so it works without a framework. For simple products, there's no need for a full React island.

## Understanding the CoCart response

The CoCart Products API returns data in this structure:

```json
{
  "id": 123,
  "name": "Nike Air Force 1",
  "slug": "nike-air-force-1",
  "type": "simple",
  "prices": {
    "price": "28000",
    "regular_price": "28000",
    "sale_price": "",
    "currency_code": "USD",
    "currency_symbol": "$",
    "currency_minor_unit": 2
  },
  "images": [
    {
      "id": 456,
      "src": "https://yourstore.com/wp-content/uploads/product.jpg",
      "alt": "Nike Air Force 1"
    }
  ],
  "attributes": [
    {
      "name": "Color",
      "slug": "pa_color",
      "options": ["Black", "White", "Red"]
    }
  ]
}
```

<Note>
  Prices are returned in the smallest currency unit (e.g. `28000` = $280.00 for USD). Use `currency_minor_unit` to determine how many decimal places to shift.
</Note>

## Handling sale prices

Show original and discounted prices when a product is on sale:

```astro
---
const { product } = Astro.props;
const { prices } = product;
const isOnSale = prices.sale_price && prices.sale_price !== prices.regular_price;

function formatPrice(price: string, symbol = '$') {
  return `${symbol}${(parseInt(price) / 100).toFixed(2)}`;
}
---

{isOnSale ? (
  <div class="flex items-center gap-2">
    <span class="text-sm text-zinc-400 line-through">
      {formatPrice(prices.regular_price, prices.currency_symbol)}
    </span>
    <span class="text-sm text-red-600 font-semibold">
      {formatPrice(prices.sale_price, prices.currency_symbol)}
    </span>
  </div>
) : (
  <span class="text-sm text-zinc-500 dark:text-zinc-300">
    {formatPrice(prices.price, prices.currency_symbol)}
  </span>
)}
```

## Customization

### Grid layouts

Change the number of columns by adjusting the grid classes:

```html
<!-- 4 columns on large screens -->
<div class="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-4">

<!-- 2 columns max -->
<div class="grid grid-cols-1 gap-6 md:grid-cols-2">
```

### Filtering by category

```typescript
// In src/lib/cocart.ts
export async function getProductsByCategory(
  categoryId: number,
  params: Record<string, string | number> = {}
) {
  return getProducts({ ...params, category: categoryId });
}
```

Then use it in your page:

```astro
---
const products = await getProductsByCategory(15, { per_page: 8 });
---
```
